<script setup>
import { ref, reactive } from 'vue'
import { ElMessage } from 'element-plus'
import { User, Lock } from '@element-plus/icons-vue'
import { useRouter } from 'vue-router'
import $API from '@/api'
 
const router = useRouter()
 
const loginFormRef = ref()

const loginLoading = ref(false)
 
const loginForm = reactive({
  username: '',
  password: '',
})

const loginRules = {
  username: [{ required: true, message: '请输入用户名', trigger: 'blur' }],
  password: [{ required: true, message: '请输入密码', trigger: 'blur' }],
}

const handleLogin = async () => {
  const valid = await loginFormRef.value.validate().catch(() => false)
 
  if (!valid) return
 
  loginLoading.value = true
 
  try {
    const res = await $API.login(loginForm)
 
    // 检查后端返回的错误状态码
    const responseCode = res.data?.code
    if (responseCode === 500 || responseCode === '500' || responseCode === 400) {
      const errorMsg = res.data?.message || res.data?.msg || '登录失败'
      throw new Error(errorMsg)
    }

    // 登录成功后，保存token和role到localStorage
    const tokenData = res.data?.data
    if (tokenData && tokenData.token) {
      localStorage.setItem('token', tokenData.token)
      localStorage.setItem('role', tokenData.role)
    } else {
      console.warn('登录响应中没有token:', res)
      throw new Error(res.data?.msg || '登录响应数据异常')
    }

    ElMessage.success('登录成功')
 
    // 根据身份跳转到对应页面
    const role = tokenData.role
    if (role === 'system_admin') {
      router.push('/index/userManage')
    } else if (role === 'enterprise_admin') {
      router.push('/index/contractorManage')
    } else {
      router.push('/index/landInfo')
    }
  } catch (error) {
    console.error('登录失败:', error)
    ElMessage.error(error.message || '登录失败')
  } finally {
    loginLoading.value = false
  }
}
</script>
